create function getEmpName(empid int)
returns varchar(100)
BEGIN
 declare ename varchar(100);
 SET ename = (select concat(p.last_name,' ',p.first_name) from employees as e inner join persons as p ON p.id = e.person_id WHERE e.id = empid LIMIT 1);
 return ename; 
END;
